    "class MyNLP:\n",
    "    def __init__(self):\n",
    "        self.dimension = 2\n",
    "        self.types = [ry.OT.sos] * 2\n",
    "        self.bounds = np.array([[-2,-1], [2,3]])\n",
    "        \n",
    "        self.b = 3\n",
    "    def evaluate(self, x):\n",
    "        phi = np.array([ x[0]-1,\n",
    "                         self.b*(x[1]-x[0]**2) ])\n",
    "        J = np.array([[ 1, 0 ],\n",
    "                      [ -2*self.b*x[0], self.b ]])\n",
    "        return phi, J\n",
    "    \n",
    "    def f(self, x):\n",
    "        phi, _ = self.evaluate(x)\n",
    "        return np.sum(phi**2)\n",
    "    \n"
   ]
